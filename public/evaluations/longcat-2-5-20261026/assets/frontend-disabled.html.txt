(function(){
  "use strict";

  /* ---------- 移动端导航 ---------- */
  var toggle = document.getElementById('navToggle');
  var links = document.getElementById('navLinks');
  toggle.addEventListener('click', function(){
    var open = links.classList.toggle('open');
    toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
    toggle.setAttribute('aria-label', open ? '关闭菜单' : '打开菜单');
  });
  links.addEventListener('click', function(e){
    if(e.target.tagName === 'A'){
      links.classList.remove('open');
      toggle.setAttribute('aria-expanded','false');
    }
  });

  /* ---------- 路线筛选 ---------- */
  var filterBtns = document.querySelectorAll('.filter-btn');
  var cards = document.querySelectorAll('.route-card');

  filterBtns.forEach(function(btn){
    btn.addEventListener('click', function(){
      var level = btn.getAttribute('data-filter');

      filterBtns.forEach(function(b){ b.setAttribute('aria-pressed','false'); });
      btn.setAttribute('aria-pressed','true');

      cards.forEach(function(card){
        var match = (level === 'all') || (card.getAttribute('data-level') === level);
        card.hidden = !match;
      });
    });
  });

  /* ---------- 报名表单 ---------- */
  var form = document.getElementById('signupForm');
  var nameInput = document.getElementById('nameInput');
  var expSelect = document.getElementById('expSelect');
  var nameField = document.getElementById('nameField');
  var expField = document.getElementById('expField');
  var successMsg = document.getElementById('successMsg');
  var successText = document.getElementById('successText');

  function validate(){
    var ok = true;
    var nameVal = nameInput.value.trim();
    var expVal = expSelect.value;

    if(nameVal === ''){
      nameField.classList.add('invalid');
      ok = false;
    } else {
      nameField.classList.remove('invalid');
    }

    if(expVal === ''){
      expField.classList.add('invalid');
      ok = false;
    } else {
      expField.classList.remove('invalid');
    }
    return ok;
  }

  nameInput.addEventListener('input', function(){
    if(nameInput.value.trim() !== '') nameField.classList.remove('invalid');
  });
  expSelect.addEventListener('change', function(){
    if(expSelect.value !== '') expField.classList.remove('invalid');
  });

  form.addEventListener('submit', function(e){
    e.preventDefault();
    if(!validate()){
      var firstInvalid = form.querySelector('.field.invalid input, .field.invalid select');
      if(firstInvalid) firstInvalid.focus();
      return;
    }

    var nameVal = nameInput.value.trim();
    var expText = expSelect.options[expSelect.selectedIndex].text;

    successText.textContent = nameVal + '，已为你登记本周骑行（' + expText + '）。请周六 09:00 前到江畔公园东门集合，带上自行车与头盔。本页面为本地演示，未发送任何数据。';
    successMsg.classList.add('show');
    successMsg.setAttribute('tabindex','-1');
    successMsg.focus();
    form.reset();
  });

})();
</script>
