<script>
(function(){
  "use strict";

  /* ---------- 路线筛选 ---------- */
  var filterBtns = Array.prototype.slice.call(document.querySelectorAll(".filter-btn"));
  var cards = Array.prototype.slice.call(document.querySelectorAll(".route-card"));
  var filterStatus = document.getElementById("filterStatus");

  filterBtns.forEach(function(btn){
    btn.addEventListener("click", function(){
      var level = btn.getAttribute("data-filter");

      filterBtns.forEach(function(b){
        b.setAttribute("aria-pressed", b === btn ? "true" : "false");
      });

      var visibleCount = 0;
      cards.forEach(function(card){
        var match = (level === "all") || (card.getAttribute("data-level") === level);
        if(match){
          card.classList.remove("is-hidden");
          card.classList.remove("is-visible");
          void card.offsetWidth; /* 重新触发动画 */
          card.classList.add("is-visible");
          visibleCount++;
        }else{
          card.classList.add("is-hidden");
          card.classList.remove("is-visible");
        }
      });

      var labelMap = { all:"全部", leisure:"休闲", advanced:"进阶" };
      filterStatus.textContent = "已显示 " + visibleCount + " 条路线（筛选：" + labelMap[level] + "）";
    });
  });

  /* ---------- 报名表单（本地演示） ---------- */
  var form = document.getElementById("signupForm");
  var nameInput = document.getElementById("nameInput");
  var expSelect = document.getElementById("expSelect");
  var nameError = document.getElementById("nameError");
  var successMsg = document.getElementById("successMsg");

  form.addEventListener("submit", function(event){
    event.preventDefault(); /* 本地演示：不导航、不发送 */

    var name = nameInput.value.trim();
    var exp = expSelect.value;

    if(!name){
      nameInput.setAttribute("aria-invalid","true");
      nameError.hidden = false;
      successMsg.hidden = true;
      nameInput.focus();
      return;
    }

    nameInput.removeAttribute("aria-invalid");
    nameError.hidden = true;
    successMsg.hidden = false;
    successMsg.textContent = "报名成功：" + name + "（" + exp + "）。这是本地演示确认，信息未离开本页面。周六见！";
    form.reset();
  });

  nameInput.addEventListener("input", function(){
    if(nameInput.value.trim()){
      nameInput.removeAttribute("aria-invalid");
      nameError.hidden = true;
    }
  });
})();
</script>
